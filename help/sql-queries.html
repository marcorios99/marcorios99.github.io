Queries SQL

Consultas reutilizables para recordar sintaxis de ventanas, agregaciones, validaciones, joins, duplicados y reportes operativos.

ROW_NUMBER para elegir el registro mas reciente

WITH ranked AS (
    SELECT
        document_id,
        status,
        updated_at,
        ROW_NUMBER() OVER (
            PARTITION BY document_id
            ORDER BY updated_at DESC
        ) AS rn
    FROM documents
)
SELECT *
FROM ranked
WHERE rn = 1;

Detectar duplicados por clave de negocio

SELECT
    business_key,
    COUNT(*) AS total_rows
FROM staging_table
GROUP BY business_key
HAVING COUNT(*) > 1;

Validar campos requeridos

SELECT *
FROM staging_table
WHERE document_id IS NULL
   OR created_at IS NULL
   OR status IS NULL;

Agregacion mensual

SELECT
    DATEFROMPARTS(YEAR(created_at), MONTH(created_at), 1) AS month_start,
    COUNT(*) AS total_documents
FROM documents
GROUP BY DATEFROMPARTS(YEAR(created_at), MONTH(created_at), 1)
ORDER BY month_start;

LEFT JOIN para encontrar faltantes

SELECT s.*
FROM source_table AS s
LEFT JOIN target_table AS t
    ON t.business_key = s.business_key
WHERE t.business_key IS NULL;

SUM OVER para acumulado

SELECT
    created_at,
    amount,
    SUM(amount) OVER (
        ORDER BY created_at
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS running_total
FROM transactions;

CASE para clasificar estados

SELECT
    document_id,
    status,
    CASE
        WHEN status IN ('APPROVED', 'VALIDATED') THEN 'ok'
        WHEN status IN ('REJECTED', 'ERROR') THEN 'review'
        ELSE 'pending'
    END AS status_group
FROM documents;

MERGE basico para upsert en SQL Server

MERGE target_table AS target
USING staging_table AS source
    ON target.business_key = source.business_key
WHEN MATCHED THEN
    UPDATE SET
        target.status = source.status,
        target.updated_at = source.updated_at
WHEN NOT MATCHED THEN
    INSERT (business_key, status, updated_at)
    VALUES (source.business_key, source.status, source.updated_at);

CTE recursivo para tablas jerarquicas

WITH hierarchy AS (
    SELECT
        id,
        parent_id,
        name,
        0 AS level,
        CAST(name AS VARCHAR(MAX)) AS path
    FROM categories
    WHERE parent_id IS NULL

    UNION ALL

    SELECT
        child.id,
        child.parent_id,
        child.name,
        parent.level + 1 AS level,
        CAST(parent.path + ' > ' + child.name AS VARCHAR(MAX)) AS path
    FROM categories AS child
    INNER JOIN hierarchy AS parent
        ON child.parent_id = parent.id
)
SELECT
    id,
    parent_id,
    name,
    level,
    path
FROM hierarchy
ORDER BY path
OPTION (MAXRECURSION 100);
